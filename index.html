/** 連続撮影画面（GitHub Pages）からの受付 */ // 撮影画面から送信できるメールアドレス（スタッフが増えたらここに追加） const ALLOWED_STAFF = [ 'info@koukado.world' ]; function doPost(e) { let res; try { const req = JSON.parse(e.postData.contents); const staff = checkStaff_(req.staff); if (req.action === 'check') res = { ok: true }; else if (req.action === 'photo') res = savePhotoFromCamera_(req, staff); else throw new Error('不明な操作です'); } catch (err) { res = { ok: false, error: err.message }; } return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON); } function checkStaff_(staff) { const s = String(staff || '').trim().toLowerCase(); const list = ALLOWED_STAFF.map(x => String(x).trim().toLowerCase()); if (list.indexOf(s) === -1) throw new Error('このメールアドレスは登録されていません（管理者に連絡してください）'); return s; } function savePhotoFromCamera_(req, staff) { const id = String(req.id || '').trim(); const index = Number(req.index); const data = String(req.image || ''); if (!/^[0-9a-f]{8}$/.test(id)) throw new Error('IDが不正です'); if (!(index >= 1 && index <= PHOTO_COLS.length)) throw new Error('写真の番号が不正です'); if (data.indexOf('data:image/jpeg;base64,') !== 0) throw new Error('画像の形式が不正です'); if (data.length > 8 * 1024 * 1024) throw new Error('画像が大きすぎます'); const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME); const lastCol = sheet.getLastColumn(); const header = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(v => String(v).trim()); const col = name => { const i = header.indexOf(name); if (i === -1) throw new Error('列「' + name + '」が見つかりません'); return i; }; const cId = col('ID'); const findRow = () => { const last = sheet.getLastRow(); if (last < 2) return -1; const ids = sheet.getRange(2, cId + 1, last - 1, 1).getValues(); for (let i = ids.length - 1; i >= 0; i--) { if (String(ids[i][0]).trim() === id) return i + 2; } return -1; }; const now = new Date(); let rowNum = findRow(); if (rowNum === -1) { const row = new Array(lastCol).fill(''); row[cId] = id; row[col('撮影日')] = Utilities.formatDate(now, TZ, 'yyyy/MM/dd'); row[col('撮影担当')] = staff; row[col('ステータス')] = '撮影済み'; row[col('登録日時')] = Utilities.formatDate(now, TZ, 'yyyy/MM/dd HH:mm:ss'); sheet.appendRow(row); SpreadsheetApp.flush(); rowNum = findRow(); if (rowNum === -1) throw new Error('商品の登録に失敗しました'); } const colName = PHOTO_COLS[index - 1]; const name = id + '.' + colName + '.' + Utilities.formatDate(now, TZ, 'HHmmss') + '.jpg'; const blob = Utilities.newBlob(Utilities.base64Decode(data.split(',')[1]), 'image/jpeg', name); const folder = getOrCreateFolder_(DriveApp.getFolderById(APP_FOLDER_ID), IMAGE_FOLDER_NAME); folder.createFile(blob); sheet.getRange(rowNum, col(colName) + 1).setValue(IMAGE_FOLDER_NAME + '/' + name); SpreadsheetApp.flush(); return { ok: true, id: id, index: index }; } finally { lock.releaseLock(); } }
